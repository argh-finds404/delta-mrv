/**
 * Parses PostGIS geometry point representations (WKT, EWKB hex, GeoJSON).
 * Guarantees [longitude, latitude] ordering.
 */
function parseGeometryPoint(value: unknown): [number, number] {
  if (!value) return [0, 0];
  if (
    Array.isArray(value) &&
    value.length === 2 &&
    typeof value[0] === "number" &&
    typeof value[1] === "number"
  ) {
    return [value[0], value[1]];
  }
  if (typeof value === "object" && value !== null && "coordinates" in value) {
    const coords = (value as { coordinates: [number, number] }).coordinates;
    if (Array.isArray(coords)) return [coords[0], coords[1]];
  }
  if (typeof value === "string") {
    const trimmed = value.trim();
    // 1. WKT: POINT(lon lat) or SRID=4326;POINT(lon lat)
    const match = trimmed.match(/POINT\s*\(\s*([-\d.]+)\s+([-\d.]+)\s*\)/i);
    if (match) {
      return [parseFloat(match[1]), parseFloat(match[2])];
    }
    // 2. EWKB Hex representation returned by standard Postgres text drivers
    if (/^[0-9a-fA-F]+$/.test(trimmed)) {
      const buf = Buffer.from(trimmed, "hex");
      if (buf.length >= 21) {
        const isLittleEndian = buf.readUInt8(0) === 1;
        const geomType = isLittleEndian
          ? buf.readUInt32LE(1)
          : buf.readUInt32BE(1);
        const hasSrid = (geomType & 0x20000000) !== 0;
        const offset = hasSrid ? 9 : 5;
        if (buf.length >= offset + 16) {
          const lon = isLittleEndian
            ? buf.readDoubleLE(offset)
            : buf.readDoubleBE(offset);
          const lat = isLittleEndian
            ? buf.readDoubleLE(offset + 8)
            : buf.readDoubleBE(offset + 8);
          return [lon, lat];
        }
      }
    }
  }
  return [0, 0];
}

/**
 * Custom PostGIS Point geometry(Point, 4326) column type per drizzle-spatial-vector-ops skill.
 */
export const geometryPoint = customType<{
  data: [number, number]; // [longitude, latitude]
  driverData: string;
}>({
  dataType() {
    return "geometry(Point, 4326)";
  },
  toDriver(value: [number, number]): string {
    const [lon, lat] = value;
    return `SRID=4326;POINT(${lon} ${lat})`;
  },
  fromDriver(value: unknown): [number, number] {
    return parseGeometryPoint(value);
  },
});
